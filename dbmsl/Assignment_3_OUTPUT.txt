create table PlacementDrive(
Drive_id int primary key auto_increment,
Pcompany_name varchar(20),
package decimal(10,2),
location varchar(20)
);

create table Training(
T_id int primary key auto_increment,
Tcompany_name varchar(20),
T_Fee decimal(10,2),
T_year year
);

create table Student(
s_id int primary key auto_increment,
Drive_id int,
T_id int,
s_name varchar(20),
CGPA decimal(3,2),
s_branch varchar(20),
S_dob date,
Foreign key (Drive_id) references PlacementDrive(Drive_id) on delete cascade,
Foreign key (T_id) references Training(T_id) on delete cascade
);

9. Create a view showing the Student and Training details.

SELECT 
    S.s_id, S.s_name, S.CGPA, S.s_branch, S.S_dob,
    T.Tcompany_name, T.T_Fee, T.T_year
FROM Student S JOIN Training T ON S.T_id = T.T_id;
